    alt not a member
        AM-->>API: 404 household.not_found

        AM-->>API: 403 auth.forbidden

        AM->>VU: checkFields({asOf: dateWithin(max: today)})

            AM-->>API: 422 validation.failed {asOf: date.max}

        alt nothing to change
            AM-->>API: the value as loaded, unaudited
        else
            AM->>IRA: update(householdId, index, valueId, changes)
            alt not_found (deleted meanwhile)
                IRA-->>AM: not_found (deleted meanwhile)

                AM-->>API: 409 index_rate.duplicate_day
            else done
                IRA-->>AM: done
                AM-->>LOG: audit EditIndexRate

            end
        end
        end
    end
